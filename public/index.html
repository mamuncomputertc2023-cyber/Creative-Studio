/**

 * File: licenseClient.js

const LICENSE_SERVER = "https://codebox-n8kk.onrender.com";

// 1. Extract unique HWID from user's machine
function getMachineHWID() {
  if (window.cep && window.cep.process) {
    // CEP Environment: Use MAC/Computer Name
    return "CEP_" + (window.navigator.userAgent.replace(/\s+/g, '') + "_" + screen.width + "x" + screen.height);
  }
  return "UXP_" + (window.navigator.userAgent + "_" + screen.width + "x" + screen.height);
}

// 2. Activate License Key
async function activatePluginLicense(licenseKey, productName = "AutoSaviour Pro") {
  const hwid = getMachineHWID();

  try {
    const response = await fetch(`${LICENSE_SERVER}/api/v1/client/activate`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        licenseKey: licenseKey.trim(),
        hwid: hwid,
        deviceName: "Illustrator-Workstation",
        appName: productName
      })
    });

    const result = await response.json();

    if (result.success && result.data.valid) {
      // Save license verification token to local storage
      localStorage.setItem("App_License_Data", JSON.stringify(result.data));
      localStorage.setItem("App_License_Sig", result.signature);
      alert(`🎉 License successfully activated! Welcome, ${result.data.customerName}`);
      return true;
    } else {
      alert(`⚠️ License verification error: ${result.message}`);
      return false;
    }
  } catch (error) {
    alert("Unable to reach license server. Please check your internet connection.");
    return false;
  }
}

// 3. Automatically check for updates on plugin startup
async function checkForPluginUpdates(productName = "AutoSaviour Pro", currentVersion = "1.2.0") {
  try {
    const url = `${LICENSE_SERVER}/api/v1/client/check-update?product=${encodeURIComponent(productName)}&currentVersion=${currentVersion}`;
    const res = await fetch(url);
    const update = await res.json();

    if (update.hasUpdate) {
      const msg = `New update available (v${update.latestVersion})!\n\nChangelog:\n${update.changelog}\n\nWould you like to download this update now?`;
      if (confirm(msg)) {
        // Open Google Drive / Cloud download link in default browser
        if (window.cep && window.cep.util) {
          window.cep.util.openURLInDefaultBrowser(update.downloadUrl);
        } else {
          window.open(update.downloadUrl);
        }
      }
    }
  } catch (err) {
    console.warn("Update check failed:", err);
  }
}

using System;
using System.IO;
using System.Net.Http;
using System.Text;
using System.Threading.Tasks;
using System.Security.Cryptography;
using System.Management; // Add Reference: System.Management.dll

public class LicenseGuard
{
    private static readonly string SERVER_URL = "https://codebox-n8kk.onrender.com";
    private static readonly HttpClient client = new HttpClient();

    // 1. Generate unique HWID from Windows CPU and Motherboard
    public static string GetWindowsHardwareID()
    {
        try
        {
            string cpuId = "";
            ManagementClass mc = new ManagementClass("Win32_Processor");
            ManagementObjectCollection moc = mc.GetInstances();
            foreach (ManagementObject mo in moc)
            {
                cpuId = mo.Properties["ProcessorId"].Value.ToString();
                break;
            }
            return "WIN_" + cpuId;
        }
        catch
        {
            return "WIN_DEV_" + Environment.MachineName;
        }
    }

    // 2. License Key Validation
    public static async Task<bool> ActivateLicenseAsync(string licenseKey, string productName)
    {
        string hwid = GetWindowsHardwareID();
        string json = $"{{\"licenseKey\":\"{licenseKey}\",\"hwid\":\"{hwid}\",\"deviceName\":\"{Environment.MachineName}\",\"appName\":\"{productName}\"}}";

        var content = new StringContent(json, Encoding.UTF8, "application/json");
        HttpResponseMessage response = await client.PostAsync($"{SERVER_URL}/api/v1/client/activate", content);
        string result = await response.Content.ReadAsStringAsync();

        if (response.IsSuccessStatusCode)
        {
            File.WriteAllText("license.key", licenseKey);
            Console.WriteLine("Software activated successfully!");
            return true;
        }
        else
        {
            Console.WriteLine("License invalid or device limit reached: " + result);
            return false;
        }
    }

    // 3. Remote Update Check
    public static async Task CheckForUpdatesAsync(string productName, string currentVersion)
    {
        string url = $"{SERVER_URL}/api/v1/client/check-update?product={Uri.EscapeDataString(productName)}&currentVersion={currentVersion}";
        string json = await client.GetStringAsync(url);
        
        if (json.Contains("\"hasUpdate\":true"))
        {
            Console.WriteLine("New update found! Opening browser download...");
            // System.Diagnostics.Process.Start(downloadUrl);
        }
    }
}

import requests
import platform
import hashlib
import uuid
import webbrowser

SERVER_URL = "https://codebox-n8kk.onrender.com"

def get_machine_hwid():
    """Generate unique HWID using user MAC address and processor architecture"""
    mac = uuid.getnode()
    raw_hwid = f"{mac}_{platform.node()}_{platform.processor()}"
    return "PY_" + hashlib.sha256(raw_hwid.encode()).hexdigest()[:16].upper()

def activate_license(license_key, product_name="My Python App"):
    hwid = get_machine_hwid()
    payload = {
        "licenseKey": license_key.strip(),
        "hwid": hwid,
        "deviceName": platform.node(),
        "appName": product_name
    }

    try:
        res = requests.post(f"{SERVER_URL}/api/v1/client/activate", json=payload, timeout=5)
        data = res.json()

        if data.get("success"):
            print(f"🎉 License activated! Customer: {data['data']['customerName']}")
            with open(".license", "w") as f:
                f.write(license_key)
            return True
        else:
            print("❌ Activation failed:", data.get("message"))
            return False
    except Exception as e:
        print("Unable to connect to license server:", e)
        return False

def check_updates(product_name="My Python App", current_version="1.0.0"):

const os = require('os');
const crypto = require('crypto');
const { shell } = require('electron');

const SERVER_URL = 'https://codebox-n8kk.onrender.com';

// 1. Generate HWID from OS CPU and Hostname
function getDeviceHWID() {
  const cpus = os.cpus();
  const raw = os.hostname() + "_" + (cpus[0] ? cpus[0].model : '') + "_" + os.platform();
  return "NODE_" + crypto.createHash('md5').update(raw).digest('hex').substring(0, 16).toUpperCase();
}

// 2. Activation Function
async function activateElectronApp(licenseKey, productName) {
  const hwid = getDeviceHWID();
  
  const response = await fetch(`${SERVER_URL}/api/v1/client/activate`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      licenseKey,
      hwid,
      deviceName: os.hostname(),
      appName: productName
    })
  });

  const data = await response.json();
  if (data.success) {
    // Save to electron-store or local secure cache
    return { success: true, customer: data.data.customerName };
  } else {
    throw new Error(data.message || 'Activation failed');
  }
}

// 3. Update Check
async function checkElectronUpdates(productName, currentVersion) {
  const res = await fetch(`${SERVER_URL}/api/v1/client/check-update?product=${encodeURIComponent(productName)}&currentVersion=${currentVersion}`);
  const update = await res.json();
  if (update.hasUpdate) {
    shell.openExternal(update.downloadUrl); // Opens Google Drive or direct download URL
  }
}
C++ / Qt Framework (Native Windows / Mac / Linux) 
#include <QCoreApplication>
#include <QNetworkAccessManager>
#include <QNetworkRequest>
#include <QNetworkReply>
#include <QJsonObject>
#include <QJsonDocument>
#include <QSysInfo>
#include <QDebug>

void activateLicense(const QString &licenseKey, const QString &productName) {
    QNetworkAccessManager *manager = new QNetworkAccessManager();
    QUrl url("https://codebox-n8kk.onrender.com/api/v1/client/activate");
    QNetworkRequest request(url);
    request.setHeader(QNetworkRequest::ContentTypeHeader, "application/json");

    QString hwid = "CPP_" + QSysInfo::machineUniqueId();

    QJsonObject json;
    json["licenseKey"] = licenseKey;
    json["hwid"] = hwid;
    json["deviceName"] = QSysInfo::machineHostName();
    json["appName"] = productName;

    QNetworkReply *reply = manager->post(request, QJsonDocument(json).toJson());

    QObject::connect(reply, &QNetworkReply::finished, [reply]() {
        if (reply->error() == QNetworkReply::NoError) {
            qDebug() << "License Validated:" << reply->readAll();
        } else {
            qDebug() << "Error:" << reply->errorString();
        }
        reply->deleteLater();
    });
}

<?php
/**
 * PHP Client License Validator (WordPress & Laravel)
 */
function validate_software_license($license_key, $product_name) {
    $server_url = 'https://codebox-n8kk.onrender.com/api/v1/client/activate';
    $domain_hwid = 'WEB_' . md5($_SERVER['SERVER_NAME'] ?? 'localhost');

    $body = json_encode([
        'licenseKey' => trim($license_key),
        'hwid' => $domain_hwid,
        'deviceName' => $_SERVER['SERVER_NAME'] ?? 'Web Server',
        'appName' => $product_name
    ]);

    $ch = curl_init($server_url);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, $body);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    
    $response = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    $data = json_decode($response, true);
    if ($http_code === 200 && isset($data['success']) && $data['success']) {
        return ['valid' => true, 'customer' => $data['data']['customerName']];
    }
    return ['valid' => false, 'message' => $data['message'] ?? 'Invalid License'];
}
